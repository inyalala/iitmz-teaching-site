def percentile(values: list[float], p: float) -> float:
    """Compute linear-interpolated p-th percentile (0 <= p <= 100)."""
    if not values:
        raise ValueError("values must be non-empty")
    s = sorted(values)
    k = (len(s) - 1) * (p / 100.0)
    f, c = math.floor(k), math.ceil(k)

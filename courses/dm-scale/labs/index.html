DM07: Recommender Systems & Matrix Factorization (SVD)

Implements user-based cosine collaborative filtering and biased latent-factor SGD matrix factorization with L2 regularization and RMSE evaluation.

Python Preview
def matrix_factorization_sgd(ratings, n_users, n_items, k=4, lr=0.03, reg=0.02, epochs=120):
    mu = float(np.mean([r for _, _, r in ratings]))
    bu, bi = np.zeros(n_users), np.zeros(n_items)
    P = rng.normal(0.0, 0.1, size=(n_users, k))
    Q = rng.normal(0.0, 0.1, size=(n_items, k))
    for _ in range(epochs):
        for u, i, r in ratings:
            err = r - (mu + bu[u] + bi[i] + float(Q[i] @ P[u]))
            bu[u] += lr * (err - reg * bu[u])
            bi[i] += lr * (err - reg * bi[i])
            P[u], Q[i] = P[u] + lr * (err * Q[i] - reg * P[u]), Q[i] + lr * (err * P[u] - reg * Q[i])
    return mu, bu, bi, P, Q

def pagerank(adj, beta=0.85, tol=1e-10, max_iter=200):
    # Dead ends redistribute uniform mass 1/n across all nodes to prevent rank leak
    r = np.full(n, 1.0 / n, dtype=float)
    teleport = (1.0 - beta) / n
    for iters in range(1, max_iter + 1):
        r_next = beta * (M @ r) + teleport
        if float(np.sum(np.abs(r_next - r))) < tol:
            return {u: float(r_next[idx[u]]) for u in nodes}, iters
        r = r_next

def flajolet_martin_estimate(stream, n_hashes=16, groups=4):
    # Track max trailing zero bit position R per hash; take median of group means
    max_r = [0] * n_hashes
    for item in stream:
        for h_idx in range(n_hashes):
            max_r[h_idx] = max(max_r[h_idx], _trailing_zeros(hash_val(item, h_idx)))
    return float(statistics.median(group_means))
